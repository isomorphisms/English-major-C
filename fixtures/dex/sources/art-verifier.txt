 5173 bool MethodVerifierImpl::VerifyFilledNewArray(const Instruction* inst, bool is_range) {
 5174   dex::TypeIndex type_idx;
 5175   if (!is_range) {
 5176     DCHECK_EQ(inst->Opcode(), Instruction::FILLED_NEW_ARRAY);
 5177     type_idx = dex::TypeIndex(inst->VRegB_35c());
 5178   } else {
 5179     DCHECK_EQ(inst->Opcode(), Instruction::FILLED_NEW_ARRAY_RANGE);
 5180     type_idx = dex::TypeIndex(inst->VRegB_3rc());
 5181   }
 5182   // Dex file verifier ensures that all valid type indexes reference valid descriptors and the
 5183   // `CheckNewArray()` ensures that the descriptor starts with an `[` before we get to the
 5184   // code flow verification. So, we should see only array types here.
 5185   const RegType& res_type = ResolveClass<CheckAccess::kYes>(type_idx);
 5186   DCHECK(res_type.IsArrayTypes());
 5187   // TODO: check Compiler::CanAccessTypeWithoutChecks returns false when res_type is unresolved
 5188   DCHECK(!res_type.IsUnresolvedMergedReference());
 5189   // Verify each input register. It's legal, if silly, for arg_count to be zero.
 5190   const RegType& expected_type = reg_types_.GetComponentType(res_type);
 5191   uint32_t arg_count = (is_range) ? inst->VRegA_3rc() : inst->VRegA_35c();
 5192   uint32_t arg[5];
 5193   if (!is_range) {
 5194     inst->GetVarArgs(arg);
 5195   }
 5196   for (size_t ui = 0; ui < arg_count; ui++) {
 5197     uint32_t get_reg = is_range ? inst->VRegC_3rc() + ui : arg[ui];
 5198     if (!VerifyRegisterType(get_reg, expected_type)) {
 5199       // Don't continue on hard failures.
 5200       DCHECK(flags_.have_pending_hard_failure_);
 5201       return false;
 5202     }
 5203     DCHECK(!flags_.have_pending_hard_failure_);
 5204   }
 5205   // filled-array result goes into "result" register
 5206   work_line_->SetResultRegisterType(res_type);
 5207   return true;
